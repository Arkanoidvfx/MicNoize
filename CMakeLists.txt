cmake_minimum_required(VERSION 3.22)
project(MicNoize LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
add_library(rubberband STATIC vendor/rubberband-4.0.0/single/RubberBandSingle.cpp)
target_include_directories(rubberband PUBLIC vendor/rubberband-4.0.0)
target_compile_definitions(rubberband PRIVATE NOMINMAX _USE_MATH_DEFINES)
add_library(mic_engine STATIC src/audio.cpp src/bridge.cpp src/tag_endpoint.cpp src/tag_task.cpp)
target_link_libraries(mic_engine PUBLIC rubberband wtsapi32 shell32)
target_include_directories(mic_engine PUBLIC src vendor/nvidia-afx-3.0.0/include vendor/nvidia-afx-3.0.0/features/nvafxdenoiser/include)
target_include_directories(mic_engine PRIVATE vendor/tag-2.0.0.1903-demo)
target_compile_definitions(mic_engine PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
# C4324 only reports Ring's intentional cache-line padding between producer and consumer.
target_compile_options(mic_engine PUBLIC /W4 /wd4324 /permissive- /utf-8 /EHsc)
target_link_libraries(mic_engine PUBLIC ole32 oleaut32 taskschd advapi32 uuid avrt mmdevapi winhttp setupapi)
add_executable(mic_check src/check.cpp)
target_include_directories(mic_check PRIVATE vendor/tag-2.0.0.1903-demo)
target_link_libraries(mic_check PRIVATE mic_engine)
target_link_options(mic_check PRIVATE /STACK:8388608)
add_executable(mic_tag src/tag_main.cpp)
target_link_libraries(mic_tag PRIVATE mic_engine)
# Explicit, offline driver probes; never run by the application or ordinary CTest.
add_executable(mic_tag_probe src/tag_probe.cpp)
target_include_directories(mic_tag_probe PRIVATE vendor/tag-2.0.0.1903-demo)
target_link_libraries(mic_tag_probe PRIVATE mic_engine)
set_target_properties(mic_tag_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin")
add_executable(mic_tag_host WIN32 src/tag_host.cpp src/tag_host.rc)
target_include_directories(mic_tag_host PRIVATE vendor/tag-2.0.0.1903-demo)
target_link_libraries(mic_tag_host PRIVATE mic_engine)
set_target_properties(mic_check mic_tag mic_tag_host PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin")
enable_testing()
add_test(NAME audio_logic COMMAND mic_check --self-test)
add_executable(effects_check src/effects_check.cpp)
target_link_libraries(effects_check PRIVATE mic_engine)
set_target_properties(effects_check PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin")
add_test(NAME voice_effects COMMAND effects_check)
add_executable(bridge_check src/bridge_check.cpp)
target_link_libraries(bridge_check PRIVATE mic_engine)
set_target_properties(bridge_check PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin")
add_test(NAME rust_bridge COMMAND bridge_check)
